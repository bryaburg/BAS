'SaStPrSptCtrl
'Built by Barbeau
'Version 2019-02-15 1427 
'Resets Static Pressure on demand

Numeric Input SaFanSts, DprPos_Max, DprPos_QtyMax, SaStPrMaxSpt, SaStPrMinSpt
Numeric Output SaStPrSpt
Numeric SaStPrResDnAmt, SaStPrResUpAmt

Init:
  SaStPrResDnAmt = .15
  SaStPrResUpAmt = .2
  Goto FanOff

FanOff:
  SaStPrSpt = SaStPrMaxSpt
  If SaFanSts is On then Goto ResetDelay

ResetDelay:
  If TS > 300 then Goto ResetCalc

ResetCalc:
  If TS > 300 then Goto ResetPr
  If SaFanSts is Off then Goto FanOff

ResetPr:
If DprPos_QtyMax >= 1 Then
  SaStPrSpt = minimum(SaStPrSpt + (DprPos_QtyMax * SaStPrResUpAmt), SaStPrMaxSpt)
Else
  If DprPos_Max > 93 Then
    SaStPrSpt = minimum(SaStPrSpt + SaStPrResUpAmt, SaStPrMaxSpt)
  Else
    If DprPos_Max < 87 Then SaStPrSpt = maximum(SaStPrSpt - SaStPrResDnAmt, SaStPrMinSpt)
  Endif
Endif
Goto ResetCalc
