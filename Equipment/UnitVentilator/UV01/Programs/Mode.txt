' ============================================================================' NOTE: SE script (EcoStruxure Building Operation) text-based language file.'       Help/syntax reference: EcoStruxure Building Operation - Plain English'       and Script Difference Guide, 04-40002-01-en (Dec 2021), Section 4'       "Program Language Differences". Native delay/wait keyword is'       "Wait <seconds>" (alias "Delay"); there is no literal "checkwait".'       System constants: True, False, On, Off. System variables: Date,'       Hour, Month, Year.' ============================================================================'' =============================================================================' HVAC Zone Control Logic' -----------------------------------------------------------------------------' Determines a single multistate zone command based on occupancy, room' temperature vs. setpoint, room relative humidity, and outside air temperature.'' Inputs:'   RmTmp           - Room temperature (°F)'   RmStpt          - Room temperature setpoint when occupied (°F)'   UnOccCoolStpt   - Cooling setpoint when unoccupied (°F)'   UnOccHeatingStpt- Heating setpoint when unoccupied (°F)'   RmRh            - Room relative humidity (%)'   OaTmp           - Outside air temperature (°F)'   Occ             - Occupancy status (0 = Unoccupied, 1 = Occupied)'' Output:'   UnitState - Multistate zone command (0-5):'     0 = Off              (fan off, nothing active)'     1 = Heating           (fan on, heat on)'     2 = Cooling           (fan on, mechanical cooling on)'     3 = Cooling + Dehum   (fan on, mechanical cooling + dehumidification on)'     4 = Economizer        (fan on, free cooling with outside air)'     5 = Fan Only          (fan on, occupied deadband - no heat/cool needed)'' Logic summary:'   - Unoccupied (Occ = 0): Only Heating(1), Cooling(2), or Off(0) - no'     economizer or dehumidification is used, and the fan runs only when'     actively heating or cooling.'       - Heating when RmTmp < UnOccHeatingStpt'       - Cooling when RmTmp > UnOccCoolStpt'       - Off when between setpoints'   - Occupied (Occ = 1): Fan runs continuously (State 5 minimum).'       - Heating when RmTmp < (RmStpt - 2°F deadband)'       - On a call for cool (RmTmp > RmStpt + 2°F), room humidity is checked:'           - If humidity is high (RmRh >= 55%) -> State 3 (Cooling + Dehum).'             Economizer is locked out here since outside air alone'             wouldn't help control humidity.'           - Otherwise, if outside air is cool enough (<= 60°F) ->'             State 4 (Economizer / free cooling).'           - If outside air is too warm -> State 2 (Cooling).'       - In the deadband (±2°F of setpoint) -> State 5 (Fan Only).'   - Any other occupancy value -> State 0 (Off), fail-safe.' =============================================================================Numeric Input RmRh, RmRhStpt, OaTmp, Occ, RmTmp, RmStpt, UnOccCoolStpt, UnOccHeatingStpt, DeadBandNumeric Output UnitStateSelect Case OccCase 0  ' UNOCCUPIED    Select Case RmTmp    Case RmTmp < UnOccHeatingStpt        UnitState = 1   ' Heating    Case RmTmp > UnOccCoolStpt        UnitState = 2   ' Cooling    Case Else        UnitState = 0   ' Off    EndSelectCase 1  ' OCCUPIED    Select Case RmTmp    Case RmTmp < (RmStpt - DeadBand)        UnitState = 1   ' Heating    Case RmTmp > (RmStpt + DeadBand)        Select Case RmRh        Case RmRh >= RmRhStpt            UnitState = 3   ' Cooling + Dehum        Case Else            Select Case OaTmp            Case OaTmp <= 60                UnitState = 4   ' Economizer            Case Else                UnitState = 2   ' Cooling            EndSelect        EndSelect    Case Else        UnitState = 5   ' Fan Only (deadband, occupied)    EndSelectCase Else  ' FAIL-SAFE    UnitState = 0   ' OffEndSelect
