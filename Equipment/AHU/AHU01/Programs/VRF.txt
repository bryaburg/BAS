' =============================================================================
' VRF Condensing Unit Command / Reversing Valve Control
' Enables the VRF compressor for cooling or heating based on zone UnitState,
' with a low-OAT lockout and a condenser alarm interlock. Reversing valve is
' de-energized for cooling, energized for heating.
'
' UnitState values: 0=Off, 1=Heating, 2=Cooling, 3=Cooling+Dehum, 4=Econ+MechClg
' =============================================================================

Numeric Input UnitState, CondAlm, OAT, SaFanSts
Numeric Output VRFCmd, RevVlvCmd



' --- Constants ---
Numeric CLG_OAT_LOCKOUT = 40    ' Minimum OAT for compressor operation [°F]
Numeric ASC_DELAY = 5           ' Anti-short-cycle delay [seconds]


' --- Unit off / idle ---
' Waits for valid call with proof of SA fan, no condenser alarm, OAT above lockout,
' and anti-short-cycle timer expired.
VRF_Off:
    VRFCmd = 0
    RevVlvCmd = 0

    ' Safety interlocks must be satisfied before any mode
    If SaFanSts = 0 Then Goto VRF_Off
    If CondAlm = 1 Then Goto VRF_Off
    If OAT < CLG_OAT_LOCKOUT Then Goto VRF_Off

    ' Mode selection (cooling and heating+dehum both run mechanical cooling)
    If UnitState = 1 Then Goto VRFHtg
    If UnitState = 2 or UnitState = 3 or UnitState = 4 Then Goto VRFClg

' --- Cooling mode ---
' Reversing valve de-energized, compressor enabled.
' Drops out on any interlock failure or mode change.
VRFClg:
    RevVlvCmd = 0

    ' Immediate safety drops
    If OAT < CLG_OAT_LOCKOUT Then Goto VRF_Off
    If SaFanSts = 0 Then Goto VRF_Off
    If CondAlm = 1 Then Goto VRF_Off

    ' Enable compressor
    VRFCmd = 1

    ' Mode change with anti-short-cycle protection
    If UnitState <> 2 and UnitState <> 3 and UnitState <> 4 and TM >= ASC_DELAY Then Goto VRF_Off

    Goto VRFClg


' --- Heating mode ---
' Reversing valve energized, compressor enabled.
' Drops out on any interlock failure or mode change.
VRFHtg:
    RevVlvCmd = 1

    ' Immediate safety drops
    If OAT < CLG_OAT_LOCKOUT Then Goto VRF_Off
    If SaFanSts = 0 Then Goto VRF_Off
    If CondAlm = 1 Then Goto VRF_Off

    ' Enable compressor
    VRFCmd = 1

    ' Mode change with anti-short-cycle protection
    If UnitState <> 1 and TM >= ASC_DELAY Then Goto VRF_Off

    Goto VRFHtg
