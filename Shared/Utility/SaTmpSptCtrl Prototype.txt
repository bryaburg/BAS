'SaTmpSptCtrl
'Built by Barbeau
'Version 2023-05-16 1701
'Controls supply temp setpoint based upon room temp

Numeric Input AhuFanSts, RmTmp, RmTmpSpt, SaTmpSpt_Min, SaTmpSpt_Max
Numeric Output SaTmpSpt, HtgClgDmd
Function PIDfct
Numeric Kp, Ki, Kd, Lerr, Intg, Bias, ResetRatio, RespTime


CalculateOff:
  SaTmpSpt = RmTmp
  HtgClgDmd = 50
  If AhuFanSts = On Then Goto SetPID

SetPID:
  RespTime = 3
  Bias = 50
  Kp = 5
  Ki = (Kp / 100) * RespTime
  Kd = Ki / 8
  Goto CalculateOn

CalculateOn:
  HtgClgDmd = PIDfct(RmTmp,RmTmpSpt,kp,ki,kd,Bias,1,Lerr,Intg)
  Goto UpdateStats

UpdateStats:
  ResetRatio = (SaTmpSpt_Max-SaTmpSpt_Min)/100
  SaTmpSpt = SaTmpSpt_Max -( ResetRatio * HtgClgDmd)
  If TS > RespTime and abs(RmTmp-RmTmpSpt) > 0.1 then Goto CalculateOn
  If AhuFanSts = Off Then Goto CalculateOff
