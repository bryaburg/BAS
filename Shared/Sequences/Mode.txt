' PURPOSE
'   Compute one multistate zone command (UnitState) from occupancy, room
'   conditions, and outdoor air.
'
' INPUTS
'   RmRh           Room relative humidity (%)
'   RmRhStpt       Occupied humidity setpoint (%)
'   OaTmp          Outside air temperature (°F)
'   Occ            Occupancy (0 = Unoccupied, 1 = Occupied)
'   RmTmp          Room temperature (°F)
'   RmStpt         Occupied temperature setpoint (°F)
'   UnOccCoolStpt  Unoccupied cooling setpoint (°F)
'   UnOccHeatingStpt Unoccupied heating setpoint (°F)
'   DeadBand       Occupied temperature deadband (°F) - must be > 0
'   OverrideMode   Manual override state 0-5 (0 = off; 1-5 forces that state)
'   EconEnable     1 = allow economizer/free cooling; 0 = lock to mechanical
'   DehumEnable    1 = allow dehumidification; 0 = mechanical cooling only
'
' OUTPUT
'   UnitState - Multistate zone command (0-5):
'     0 = Off           1 = Heating
'     2 = Cooling       3 = Cooling + Dehum
'     4 = Economizer    5 = Fan Only (occupied deadband)
'
' LOGIC SUMMARY
'   Occupied (Occ = 1): fan runs continuously (State 5 minimum).
'     - RmTmp < RmStpt - DeadBand  -> Heating (1)
'     - Above the band:
'         RmRh >= RmRhStpt and DehumEnable -> Cooling + Dehum (3)
'         otherwise OaTmp <= 60 and EconEnable -> Economizer (4)
'         otherwise -> Cooling (2)
'     - In the band -> Fan Only (5)
'   Unoccupied (Occ = 0): only Heating/Cooling/Off; no economizer or dehum.
'   Any other occupancy value: fail-safe Off.
'

Numeric Input RmRh, OaTmp, RmTmp,
Numeric Public RmStpt, RmRhStpt, Occ, UnOccCoolStpt, UnOccHeatingStpt, DeadBand, EconEnable, DehumEnable, ValidInputs
Numeric Output UnitState

' ============================================================================
' REFACTORED: Select Case converted to nested If statements.
' Flow: CheckError (validation) -> ModeSelection (logic) -> CheckError (revalidate)
' ============================================================================

CheckError:
    ' Initialize validation flag
    ValidInputs = 1

    ' SAFETY CHECKS: Validate all input ranges first
    If OaTmp < -40 Or OaTmp > 150 Then
        UnitState = 0
        ValidInputs = 0
    EndIf

    If RmTmp < -40 Or RmTmp > 150 Then
        UnitState = 0
        ValidInputs = 0
    EndIf

    If RmRh < 0 Or RmRh > 100 Then
        UnitState = 0
        ValidInputs = 0
    EndIf

    ' Set safe defaults for invalid inputs
    If RmStpt < 40 Or RmStpt > 90 Then
        RmStpt = 72
    EndIf

    If UnOccHeatingStpt > UnOccCoolStpt Then
        UnOccCoolStpt = 80
        UnOccHeatingStpt = 60
    EndIf

    If RmRhStpt < 0 Or RmRhStpt > 100 Then
        RmRhStpt = 65
    EndIf

    If EconEnable < 0 Or EconEnable > 1 Then
        EconEnable = 0
    EndIf

    If DehumEnable < 0 Or DehumEnable > 1 Then
        DehumEnable = 0
    EndIf

    If Occ < 0 Or Occ > 1 Then
        Occ = 0
    EndIf

    ' After all safety checks, proceed to mode selection if valid inputs
    If ValidInputs = 1 Then
        Goto ModeSelection
    Else
        UnitState = 0   ' Off if invalid inputs
    EndIf

ModeSelection:
    ' -------------------------------------------------------------------
    ' OCCUPIED MODE (Occ = 1)
    ' Fan runs continuously (State 5 minimum when in deadband)
    ' -------------------------------------------------------------------
    If Occ = 1 Then
        ' HEATING: Below occupied deadband
        If RmTmp < (RmStpt - DeadBand) Then
            UnitState = 1   ' Heating

        ' COOLING: Above occupied deadband
        ElseIf RmTmp > (RmStpt + DeadBand) Then
            ' DEHUMIDIFICATION CHECK
            If RmRh >= RmRhStpt Then
                If DehumEnable Then
                    UnitState = 3   ' Cooling + Dehum
                Else
                    UnitState = 2   ' Cooling
                EndIf
            ' ECONOMIZER CHECK
            Else
                If OaTmp <= 60 And EconEnable Then
                    UnitState = 4   ' Economizer
                Else
                    UnitState = 2   ' Cooling
                EndIf
            EndIf

        ' FAN ONLY: In the band
        Else
            UnitState = 5   ' Fan Only (occupied deadband)
        EndIf

        ' After occupied mode selected, return to CheckError for revalidation
        Goto CheckError

    ' -------------------------------------------------------------------
    ' UNOCCUPIED MODE (Occ = 0)
    ' Only Heating/Cooling/Off; no economizer or dehum
    ' -------------------------------------------------------------------
    ElseIf Occ = 0 Then
        ' HEATING: Below unoccupied heating setpoint
        If RmTmp < UnOccHeatingStpt Then
            UnitState = 1   ' Heating

        ' COOLING: Above unoccupied cooling setpoint
        ElseIf RmTmp > UnOccCoolStpt Then
            UnitState = 2   ' Cooling

        ' OFF: In between setpoints
        Else
            UnitState = 0   ' Off
        EndIf

        ' After unoccupied mode selected, return to CheckError for revalidation
        Goto CheckError

    ' -------------------------------------------------------------------
    ' FAIL-SAFE: Any other occupancy value
    ' -------------------------------------------------------------------
    Else
        UnitState = 0   ' Off (fail-safe)
    EndIf

    ' Return to CheckError for input validation on next cycle
    Goto CheckError
